import numpy as np
import matplotlib.pyplot as plt


def compute_pi(n):
    if n == 0:
        return 0.0
    return n * np.sin(np.pi / n)


def compute_error(pi_approx, pi_exact=np.pi):
    return np.abs(pi_exact - pi_approx)


n_list = [1, 2, 4, 8, 16, 32, 64, 128, 256]
pi_approx_list = []
error_list = []
h_list = []

print("=" * 70)
print(f"{'n':<6} {'π_n (近似值)':<25} {'误差 e_n':<20} {'h=1/n':<10}")
print("=" * 70)
for n in n_list:
    pi_n = compute_pi(n)
    e_n = compute_error(pi_n)
    h = 1.0 / n

    pi_approx_list.append(pi_n)
    error_list.append(e_n)
    h_list.append(h)

    print(f"{n:<6} {pi_n:<25.15f} {e_n:<20.2e} {h:<10.4f}")

pi_exact = np.pi
print("\n真实π值（高精度）: ", pi_exact)

log_h = np.log10(h_list[1:])
log_e = np.log10(error_list[1:])
slope, intercept = np.polyfit(log_h, log_e, 1)
print(f"\n拟合误差的收敛率（log-log斜率）: {slope:.2f}（理论值为2）")

plt.figure(figsize=(8, 6))
plt.loglog(h_list, error_list, 'o-', color='#1f77b4', linewidth=1.5, label=r'$e_n = |\pi - \pi_n|$')

h_theory = np.logspace(-3, 0, 100)
e_theory = (np.pi ** 3 / 6) * h_theory ** 2
plt.loglog(h_theory, e_theory, '--', color='#ff7f0e', linewidth=1.2, label=r'理论斜率 $O(h^2)$')

plt.xlabel(r'$h = 1/n$ (单元尺寸)', fontsize=12)
plt.ylabel(r'$e_n = |\pi - \pi_n|$ (误差)', fontsize=12)
plt.title('多边形逼近π的误差收敛曲线', fontsize=14)
plt.grid(True, which="both", ls="--", alpha=0.5)
plt.legend(fontsize=12)
plt.show()
